/* ACO-63 GENESIS R0.5 — experimental durable relay / Cloudflare Worker + D1.
 * No key custody, trading, on-chain signing or financial APIs.
 * Does NOT verify MetaMask signatures: XR client does that before acting.
 * Not a public multi-user production authentication service.
 */
const ALLOWED_ORIGIN = 'https://apollom911.github.io';
const roomRe = /^[0-9a-f]{20}$/;
const secretRe = /^[0-9a-f]{64}$/;
const idRe = /^batch-[0-9a-f]{16}$/;
const hexRe = /^[0-9a-f]{32}$/;
const walletRe = /^0x[0-9a-fA-F]{40}$/;
const signatureRe = /^0x[0-9a-fA-F]{130}$/;
const encoder = new TextEncoder();
const now = () => Date.now();
function resp(value, status=200, origin=null) {
 const headers = {'content-type':'application/json; charset=utf-8','cache-control':'no-store','x-content-type-options':'nosniff'};
 if (origin === ALLOWED_ORIGIN) {
  headers['access-control-allow-origin']=ALLOWED_ORIGIN;
  headers['access-control-allow-methods']='GET, POST, OPTIONS';
  headers['access-control-allow-headers']='content-type, authorization, x-aco-setup';
  headers['vary']='Origin';
 }
 return new Response(JSON.stringify(value),{status,headers});
}
function err(status, code, origin) { return resp({ok:false,error:code}, status, origin); }
async function hexSha(str) {return [...new Uint8Array(await crypto.subtle.digest('SHA-256',encoder.encode(str)))].map(b=>b.toString(16).padStart(2,'0')).join('');}
function same(a,b){if(typeof a !== 'string'||typeof b !== 'string'||a.length!==b.length)return false;let out=0;for(let i=0;i<a.length;i++)out|=a.charCodeAt(i)^b.charCodeAt(i);return out===0;}
let schemaDatabase = null;
async function schema(db){
 if (schemaDatabase === db) return;
 await db.batch([
  db.prepare('CREATE TABLE IF NOT EXISTS rooms(id TEXT PRIMARY KEY, secret_hash TEXT NOT NULL, owner_wallet TEXT NOT NULL, created_ms INTEGER NOT NULL, expires_ms INTEGER NOT NULL)'),
  db.prepare('CREATE TABLE IF NOT EXISTS batches(id TEXT PRIMARY KEY, room_id TEXT NOT NULL, wallet TEXT NOT NULL, nonce TEXT NOT NULL, seed TEXT NOT NULL, job_count INTEGER NOT NULL, issued_ms INTEGER NOT NULL, expires_ms INTEGER NOT NULL, signature TEXT NOT NULL, created_ms INTEGER NOT NULL, UNIQUE(room_id, nonce))'),
  db.prepare('CREATE TABLE IF NOT EXISTS receipts(batch_id TEXT NOT NULL, job_index INTEGER NOT NULL, digest_hex TEXT NOT NULL, worker_node TEXT NOT NULL, completed_ms INTEGER NOT NULL, created_ms INTEGER NOT NULL, PRIMARY KEY(batch_id,job_index))')
 ]);
 schemaDatabase = db;
}
async function bodyJson(req){const sz=Number(req.headers.get('content-length')||0);if(sz>18000)throw new Error('PAYLOAD_TOO_LARGE');const txt=await req.text();if(txt.length>18000)throw new Error('PAYLOAD_TOO_LARGE');return JSON.parse(txt);}
async function access(req,db,roomId){
 const auth=req.headers.get('authorization')||'';
 const token=auth.startsWith('Bearer ')?auth.slice(7):'';
 if(!roomRe.test(roomId)||!secretRe.test(token))return null;
 const room=await db.prepare('SELECT * FROM rooms WHERE id=?').bind(roomId).first();
 if(!room||room.expires_ms<=now())return null;
 if(!same(room.secret_hash,await hexSha(token)))return null;
 return room;
}
async function api(req,env){
 const origin=req.headers.get('origin');
 const url=new URL(req.url);
 if(req.method==='OPTIONS')return origin===ALLOWED_ORIGIN?resp({ok:true},200,origin):err(403,'ORIGIN_NOT_ALLOWED',origin);
 if(url.pathname==='/autonomy' && req.method==='GET'){
  if (!env.DB) return err(503,'MISSING_D1_BINDING',origin);
  await autonomySchema(env.DB);
  const latest=await env.DB.prepare('SELECT slot_ms,occurred_ms,digest_hex,jobs_added FROM autonomy_pulses ORDER BY slot_ms DESC LIMIT 1').first();
  const count=await env.DB.prepare('SELECT COUNT(*) AS n FROM receipts WHERE worker_node=?').bind('edge-r051').first();
  return resp({ok:true,schema:'aco63-relay/0.5.1',cron_configured:'check Cloudflare Triggers separately',last_pulse:latest||null,edge_jobs_recorded:count?.n||0,financial_execution:false,scope:'pilot SHA-256 only'},200,origin);
 }
 if(url.pathname==='/health' && req.method==='GET'){if(env.DB && env.SETUP_KEY)await schema(env.DB);return resp({ok:true,schema:'aco63-relay/0.5',storage:!!env.DB,setup_key_configured:!!env.SETUP_KEY,financial_execution:false,security:'pilot only'},200,origin);}
 if(!env.DB||!env.SETUP_KEY)return err(503,'MISSING_D1_BINDING_OR_SETUP_KEY',origin);
 if(origin!==ALLOWED_ORIGIN)return err(403,'ORIGIN_NOT_ALLOWED',origin);
 if(!['GET','POST'].includes(req.method))return err(405,'METHOD_NOT_ALLOWED',origin);
 // Tables created lazily to avoid running a separate SQL console step.
 await schema(env.DB);
 if(url.pathname==='/room' && req.method==='POST'){
  if(!same(req.headers.get('x-aco-setup')||'',env.SETUP_KEY))return err(401,'SETUP_KEY_INVALID',origin);
  const b=await bodyJson(req);
  if(!roomRe.test(b.id||'')||!secretRe.test(b.secret||'')||!walletRe.test(b.wallet||''))return err(400,'INVALID_ROOM_DETAILS',origin);
  const existing=await env.DB.prepare('SELECT id FROM rooms WHERE id=?').bind(b.id).first();
  if(existing)return err(409,'ROOM_EXISTS',origin);
  const t=now();
  await env.DB.prepare('INSERT INTO rooms(id,secret_hash,owner_wallet,created_ms,expires_ms) VALUES (?,?,?,?,?)').bind(b.id,await hexSha(b.secret),b.wallet.toLowerCase(),t,t+48*3600*1000).run();
  return resp({ok:true,id:b.id,expires_ms:t+48*3600*1000,warning:'Room secret is not a financial credential; browser pilot only.'},201,origin);
 }
 const roomId=req.headers.get('x-aco-room')||url.searchParams.get('room')||'';
 const room=await access(req,env.DB,roomId);
 if(!room)return err(401,'ROOM_ACCESS_DENIED',origin);
 if(url.pathname==='/batch' && req.method==='POST'){
  const b=await bodyJson(req);const t=now();
  if(!idRe.test(b.id||'')||!hexRe.test(b.nonce||'')||!hexRe.test(b.seed||'')||!walletRe.test(b.wallet||'')||!signatureRe.test(b.signature||''))return err(400,'INVALID_BATCH',origin);
  if(b.wallet.toLowerCase()!==room.owner_wallet)return err(403,'WALLET_MISMATCH',origin);
  if(!Number.isInteger(b.job_count)||b.job_count<1||b.job_count>30)return err(400,'JOB_COUNT_RANGE',origin);
  if(!Number.isInteger(b.issued_ms)||!Number.isInteger(b.expires_ms)||Math.abs(t-b.issued_ms)>300000||b.expires_ms<=t||b.expires_ms>t+12*3600*1000)return err(400,'INVALID_TIME_WINDOW',origin);
  const num=await env.DB.prepare('SELECT COUNT(*) n FROM batches WHERE room_id=?').bind(roomId).first();
  if(num.n>=3)return err(429,'PILOT_MAX_THREE_BATCHES_PER_ROOM',origin);
  try{
   await env.DB.prepare('INSERT INTO batches(id,room_id,wallet,nonce,seed,job_count,issued_ms,expires_ms,signature,created_ms) VALUES (?,?,?,?,?,?,?,?,?,?)')
    .bind(b.id,roomId,room.owner_wallet,b.nonce,b.seed,b.job_count,b.issued_ms,b.expires_ms,b.signature,t).run();
  }catch(e){return err(409,'DUPLICATE_BATCH_OR_NONCE',origin)}
  return resp({ok:true,batch_id:b.id,queued:b.job_count,server_signature_verification:false},201,origin);
 }
 if(url.pathname==='/batches' && req.method==='GET'){
  const r=await env.DB.prepare('SELECT b.* FROM batches b WHERE b.room_id=? ORDER BY b.created_ms DESC LIMIT 3').bind(roomId).all();
  return resp({ok:true,batches:r.results.map(b=>({id:b.id,room_id:b.room_id,wallet:b.wallet,nonce:b.nonce,seed:b.seed,job_count:b.job_count,issued_ms:b.issued_ms,expires_ms:b.expires_ms,signature:b.signature}))},200,origin);
 }
 if(url.pathname==='/receipt' && req.method==='POST'){
  const b=await bodyJson(req);
  if(!idRe.test(b.batch_id||'')||!Number.isInteger(b.job_index)||!/^xr-[0-9a-f]{12}$/.test(b.worker_node||'')||!Number.isInteger(b.completed_ms)||!/^([0-9a-f]{64})$/.test(b.digest_hex||''))return err(400,'INVALID_RECEIPT',origin);
  const batch=await env.DB.prepare('SELECT * FROM batches WHERE id=? AND room_id=?').bind(b.batch_id,roomId).first();
  if(!batch||batch.expires_ms<now()||b.job_index<1||b.job_index>batch.job_count)return err(403,'BATCH_UNAVAILABLE_OR_EXPIRED',origin);
  const expected=await hexSha(`ACO63-R0.5|${batch.seed}|${b.job_index}`);
  if(!same(expected,b.digest_hex))return err(422,'DIGEST_MISMATCH',origin);
  const result=await env.DB.prepare('INSERT OR IGNORE INTO receipts(batch_id,job_index,digest_hex,worker_node,completed_ms,created_ms) VALUES (?,?,?,?,?,?)').bind(batch.id,b.job_index,b.digest_hex,b.worker_node,b.completed_ms,now()).run();
  return resp({ok:true,recorded:result.meta?.changes===1,already_present:result.meta?.changes===0,batch_id:batch.id,job_index:b.job_index},200,origin);
 }
 if(url.pathname==='/status' && req.method==='GET'){
  const b=await env.DB.prepare('SELECT id,job_count,expires_ms FROM batches WHERE room_id=? ORDER BY created_ms DESC LIMIT 3').bind(roomId).all();
  const reports=[];
  for(const x of b.results){
   const rec=await env.DB.prepare('SELECT job_index,digest_hex,worker_node,completed_ms FROM receipts WHERE batch_id=? ORDER BY job_index').bind(x.id).all();
   reports.push({id:x.id,total:x.job_count,completed:rec.results.length,expires_ms:x.expires_ms,receipts:rec.results});
  }
  return resp({ok:true,room_id:roomId,batches:reports,server_time_ms:now()},200,origin);
 }
 return err(404,'NOT_FOUND',origin);
}

// R0.5.1: bounded, unattended CLOUD-SIDE demo execution.
// This is NOT XR hardware execution, not economic autonomy, and not server-side wallet signature verification.
// Cron must be configured separately in Cloudflare to run automatically.
async function autonomySchema(db) {
 await db.prepare('CREATE TABLE IF NOT EXISTS autonomy_pulses(slot_ms INTEGER PRIMARY KEY, occurred_ms INTEGER NOT NULL, digest_hex TEXT NOT NULL, jobs_added INTEGER NOT NULL)').run();
}
async function autonomyTick(env) {
 if (!env.DB || !env.SETUP_KEY) throw new Error('RELAY_NOT_CONFIGURED');
 await schema(env.DB);
 await autonomySchema(env.DB);
 const t = now();
 const slot = Math.floor(t / 300000) * 300000;
 let added = 0;
 const rows = await env.DB.prepare('SELECT b.id,b.seed,b.job_count FROM batches b JOIN rooms r ON b.room_id=r.id WHERE b.expires_ms>? AND r.expires_ms>? ORDER BY b.created_ms ASC LIMIT 3').bind(t,t).all();
 for (const batch of rows.results) {
  const n = Math.min(30, Math.max(0, Number(batch.job_count) || 0));
  for (let i=1; i<=n && added<30; i++) {
   const digest = await hexSha(`ACO63-R0.5|${batch.seed}|${i}`);
   const write = await env.DB.prepare('INSERT OR IGNORE INTO receipts(batch_id,job_index,digest_hex,worker_node,completed_ms,created_ms) VALUES(?,?,?,?,?,?)').bind(batch.id,i,digest,'edge-r051',t,now()).run();
   if (write.meta?.changes===1) added++;
  }
  if (added>=30) break;
 }
 const proof=await hexSha(`ACO63-R0.5.1|SCHEDULED-COMPUTE|${slot}`);
 await env.DB.prepare('INSERT OR IGNORE INTO autonomy_pulses(slot_ms,occurred_ms,digest_hex,jobs_added) VALUES (?,?,?,?)').bind(slot,now(),proof,added).run();
 console.log('ACO63_SCHEDULED_PULSE',JSON.stringify({slot_ms:slot,completed_jobs:added}));
}

export default {
 async scheduled(controller,env,ctx){ await autonomyTick(env); },
 async fetch(request,env){
 const origin=request.headers.get('origin');
 try{return await api(request,env)}
 catch(e){console.error('ACO relay error',e);return err(500,'INTERNAL_ERROR',origin)}
}};
